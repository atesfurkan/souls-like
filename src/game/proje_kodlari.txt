#[derive(Component)]
pub struct Stamina {
    pub current: f32,
    pub max: f32,
    pub regen_rate: f32,
    pub sprint_cost: f32,
    pub jump_cost: f32,
    pub regen_timer: Timer,
}

impl Default for Stamina {
    fn default() -> Self {
        Self {
            current: 100.0,
            max: 100.0,
            regen_rate: 20.0,
            sprint_cost: 25.0, // Saniyede harcanan stamina
            jump_cost: 10.0, // Saniyede harcanan stamina
            // 1.5 saniyelik bekleme süresi, sürekli baştan kurulacak
            regen_timer: Timer::from_seconds(1.5, TimerMode::Once),
        }
    }
}

impl Stamina {
    // Hem anlık (attack) hem sürekli (sprint) tüketim için ortak metot
    pub fn try_drain(&mut self, amount: f32) -> bool {
        if self.current > 0.0 {
            self.current -= amount;
            if self.current < 0.0 {
                self.current = 0.0;
            }
            self.regen_timer.reset(); // Harcama yapıldığı an yenilenme sayacını başa sar
            true
        } else {
            false
        }
    }
}

pub fn stamina_regen(time: Res<Time>, mut query: Query<&mut Stamina>) {
    for mut stamina in query.iter_mut() {
        stamina.regen_timer.tick(time.delta());
        
        if stamina.regen_timer.finished() && stamina.current < stamina.max {
            stamina.current += stamina.regen_rate * time.delta_seconds();
            if stamina.current > stamina.max {
                stamina.current = stamina.max;
            }
        }
    }
}

use bevy::prelude::*;
use bevy_rapier3d::prelude::*;
use super::components::{DodgeConfig, DodgeState};
use crate::game::player::components::Player;
use crate::game::common::components::MainCamera;
use crate::game::stamina::components::Stamina;

pub fn dodge_input(
    mut commands: Commands,
    keys: Res<ButtonInput<KeyCode>>,
    config: Res<DodgeConfig>,
    // Karakter zaten dodge atıyorsa tekrar atamaması için Without<DodgeState> kullanıyoruz
    mut query: Query<(Entity, &mut Stamina, &Transform), (With<Player>, Without<DodgeState>)>,
    camera_query: Query<&Transform, (With<MainCamera>, Without<Player>)>,
) {
    // Sol Alt tuşu ile dodge (İsteğe göre değiştirebilirsin)
    if !keys.just_pressed(KeyCode::AltLeft) { return; }

    let Ok((entity, mut stamina, player_transform)) = query.get_single_mut() else { return };
    let Ok(camera_transform) = camera_query.get_single() else { return };

    // Yeterli stamina varsa kes ve işleme başla
    if stamina.try_drain(config.stamina_cost) {
        let mut direction = Vec3::ZERO;

        let forward = camera_transform.forward();
        let forward_flat = Vec3::new(forward.x, 0.0, forward.z).normalize_or_zero();
        let right = camera_transform.right();
        let right_flat = Vec3::new(right.x, 0.0, right.z).normalize_or_zero();

        if keys.pressed(KeyCode::KeyW) { direction += forward_flat; }
        if keys.pressed(KeyCode::KeyS) { direction -= forward_flat; }
        if keys.pressed(KeyCode::KeyD) { direction += right_flat; }
        if keys.pressed(KeyCode::KeyA) { direction -= right_flat; }

        // Eğer hiçbir tuşa basılmıyorsa, karakterin mevcut baktığı yöne (ileri) atıl
        if direction == Vec3::ZERO {
            let player_forward = player_transform.forward();
            direction = Vec3::new(player_forward.x, 0.0, player_forward.z).normalize_or_zero();
        } else {
            direction = direction.normalize_or_zero();
        }

        // Oyuncuya Dodge State'i ekle (Dodge işlemini başlat)
        commands.entity(entity).insert(DodgeState {
            direction,
            timer: Timer::from_seconds(config.duration, TimerMode::Once),
        });
    }
}

pub fn dodge_movement(
    mut commands: Commands,
    time: Res<Time>,
    config: Res<DodgeConfig>,
    mut query: Query<(Entity, &mut DodgeState, &mut KinematicCharacterController, &mut Transform)>,
) {
    for (entity, mut state, mut controller, mut transform) in query.iter_mut() {
        state.timer.tick(time.delta());

        if state.timer.finished() {
            // Dodge süresi doldu, State'i sil ve oyuncuyu serbest bırak
            commands.entity(entity).remove::<DodgeState>();
        } else {
            // Dodge süresince karakteri kilitli yöne doğru fırlat
            let mut movement = state.direction * config.speed * time.delta_seconds();
            movement.y = -9.8 * time.delta_seconds(); // Havaya uçmasın diye yerçekimi
            
            controller.translation = Some(movement);

            // Karakter dodge attığı yöne baksın
            let target_pos = transform.translation + state.direction;
            transform.look_at(target_pos, Vec3::Y);
        }
    }
}

pub fn process_damage(
    mut events: EventReader<DamageEvent>,
    mut health_query: Query<&mut Health>,
) {
    for ev in events.read() {
        // Eğer hasar alan entity'nin Health componenti varsa canını düşür
        if let Ok(mut health) = health_query.get_mut(ev.target) {
            health.current -= ev.amount;
        }
    }
}

pub fn check_death(
    mut commands: Commands,
    query: Query<(Entity, &Health)>,
) {
    for (entity, health) in query.iter() {
        if health.current <= 0.0 {
            // Canı sıfırlanan entity'i ve tüm çocuklarını (varsa silahı, zırhı vs) dünyadan sil
            commands.entity(entity).despawn_recursive();
        }
    }
}

pub fn spawn_world(
    mut commands: Commands,
    mut meshes: ResMut<Assets<Mesh>>,
    mut materials: ResMut<Assets<StandardMaterial>>,
) {
    // Görsel zemin (Plane)
    commands.spawn((
        PbrBundle {
            mesh: meshes.add(Plane3d::default().mesh().size(50.0, 50.0)),
            material: materials.add(Color::srgb(0.3, 0.4, 0.3)),
            ..default()
        },

            illuminance: 10000.0,
            shadows_enabled: true,
            ..default()
        },
        transform: Transform::from_rotation(Quat::from_rotation_x(-std::f32::consts::FRAC_PI_4)),
        ..default()
    });

    // Rastgele Engeller (Obstacles)
    let mut rng = rand::thread_rng();
    let cube_mesh = meshes.add(Cuboid::new(2.0, 2.0, 2.0));
    let cube_mat = materials.add(Color::srgb(0.4, 0.4, 0.4));

    for _ in 0..20 {
        let x: f32 = rng.gen_range(-20.0..20.0);
        let z: f32 = rng.gen_range(-20.0..20.0);

        // Oyuncunun tam merkezde (0, 0, 0) doğduğunu biliyoruz.
        // O yüzden merkeze 3 birimden yakınsa küp oluşturmayı atla ki oyuncu taşın içinde sıkışmasın.
        if x > -3.0 && x < 3.0 && z > -3.0 && z < 3.0 {
            continue;
        }

        commands.spawn((
            PbrBundle {
                mesh: cube_mesh.clone(),
                material: cube_mat.clone(),
                transform: Transform::from_xyz(x, 1.0, z),
                ..default()
            },
            // Engel Fizikleri
            RigidBody::Fixed,
            Collider::cuboid(1.0, 1.0, 1.0), // 2x2x2 küp için yarı boyutları (half-extents)
        ));
    }
}

pub fn player_movement(
    keys: Res<ButtonInput<KeyCode>>,
    time: Res<Time>,
    config: Res<PlayerMovementConfig>, // EKLENDİ: Magic number'lar yerine Config
    mut player_query: Query<(
        &mut Transform, 
        &mut KinematicCharacterController,
        Option<&KinematicCharacterControllerOutput>, // EKLENDİ: Yere değme kontrolü
        &mut PlayerState, // EKLENDİ: Dikey hız hafızası
        &mut Stamina,
    ), (With<Player>, Without<DodgeState>)>,
    camera_query: Query<&Transform, (With<MainCamera>, Without<Player>)>,
) {
    let Ok((mut player_transform, mut controller, output, mut state, mut stamina)) = player_query.get_single_mut() else { return };
    let Ok(camera_transform) = camera_query.get_single() else { return };

    let mut direction = Vec3::ZERO;

    let forward = camera_transform.forward();
    let forward_flat = Vec3::new(forward.x, 0.0, forward.z).normalize_or_zero();
    let right = camera_transform.right();
    let right_flat = Vec3::new(right.x, 0.0, right.z).normalize_or_zero();

    if keys.pressed(KeyCode::KeyW) { direction += forward_flat; }
    if keys.pressed(KeyCode::KeyS) { direction -= forward_flat; }
    if keys.pressed(KeyCode::KeyD) { direction += right_flat; }
    if keys.pressed(KeyCode::KeyA) { direction -= right_flat; }

    // Sprint Mantığı: Shift'e basılıysa sprint hızını, değilse yürüme hızını al
    // --- SPRINT MANTIĞI GÜNCELLENDİ ---
    let mut speed = config.walk_speed;
    
    // Karakter hareket ediyorsa ve Shift'e basılıysa
    if keys.pressed(KeyCode::ShiftLeft) && direction != Vec3::ZERO {
        let cost = stamina.sprint_cost * time.delta_seconds();
        
        // Staminada yer varsa sprint yap ve staminayı tüket
        if stamina.try_drain(cost) {
            speed = config.sprint_speed;
        }
    }
    let mut movement = direction.normalize_or_zero() * speed * time.delta_seconds();

    // Zıplama ve Yerçekimi Mantığı
    // Rapier çıktısı henüz oluşmadıysa (ilk frame) false kabul et
    let is_grounded = match output {
        Some(out) => out.grounded,
        None => false,
    };

    if is_grounded {
        // Yerdeysek hafif eksi hız vererek zemine yapışık kalmasını sağla
        state.vertical_velocity = -0.1; 
        
        // Sadece yerdeyken Space tuşuna basıldığında zıpla
        if keys.just_pressed(KeyCode::Space) {
            let cost = stamina.jump_cost ;
            state.vertical_velocity = config.jump_velocity;
            stamina.try_drain(cost);
        }
    } else {
        // Havadaysak yerçekimini dikey hıza uygula (ivmelenme)
        state.vertical_velocity += config.gravity * time.delta_seconds();
    }

    // Dikey hızı (Y ekseni) asıl harekete dahil et
    movement.y = state.vertical_velocity * time.delta_seconds();

    controller.translation = Some(movement);

    if direction != Vec3::ZERO {
        let target_position = player_transform.translation + direction;
        player_transform.look_at(target_position, Vec3::Y);
    }
}

pub fn cursor_management(
    mut windows: Query<&mut Window, With<PrimaryWindow>>,
    mouse: Res<ButtonInput<MouseButton>>,
    keys: Res<ButtonInput<KeyCode>>,
) {
    let Ok(mut window) = windows.get_single_mut() else { return };
    
    // Ekrana tıklayınca fareyi kilitle
    if mouse.just_pressed(MouseButton::Left) {
        window.cursor.grab_mode = CursorGrabMode::Locked;
        window.cursor.visible = false;
    }
    // ESC'ye basınca fareyi serbest bırak
    if keys.just_pressed(KeyCode::Escape) {
        window.cursor.grab_mode = CursorGrabMode::None;
        window.cursor.visible = true;
    }
}

pub fn camera_controls(
    mut ev_motion: EventReader<MouseMotion>,
    mut camera_query: Query<(&mut Transform, &mut MainCamera)>,
    player_query: Query<&Transform, (With<Player>, Without<MainCamera>)>,
    windows: Query<&Window, With<PrimaryWindow>>,
) {
    let Ok(window) = windows.get_single() else { return };
    let Ok(player_transform) = player_query.get_single() else { return };
    let Ok((mut cam_transform, mut cam_state)) = camera_query.get_single_mut() else { return };

    // Sadece fare kilitliyken kamerayı döndür
    if window.cursor.grab_mode != CursorGrabMode::Locked {
        return;
    }

    let mut mouse_delta = Vec2::ZERO;
    for ev in ev_motion.read() {
        mouse_delta += ev.delta;
    }

    let sensitivity = 0.003;
    cam_state.yaw -= mouse_delta.x * sensitivity;
    cam_state.pitch -= mouse_delta.y * sensitivity;

    // Pitch (yukarı/aşağı) açısını sınırla, kameranın ters dönmesini engelle
    cam_state.pitch = cam_state.pitch.clamp(-1.0, 1.5);

    // Yeni dönüş rotasyonunu hesapla (Önce Yaw, sonra Pitch)
    let rotation = Quat::from_axis_angle(Vec3::Y, cam_state.yaw)
        * Quat::from_axis_angle(Vec3::X, cam_state.pitch);

    // Oyuncunun pozisyonuna offset (mesafe) ekle
    let target_offset = rotation * Vec3::new(0.0, 0.0, cam_state.distance);
    
    // Kameranın hedefini oyuncunun biraz yukarısı (gövdesi/kafası) olarak ayarla
    let look_target = player_transform.translation + Vec3::new(0.0, 1.0, 0.0);
    
    cam_transform.translation = look_target + target_offset;
    cam_transform.look_at(look_target, Vec3::Y);
}
