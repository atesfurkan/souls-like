/// 3D koordinatları karma (hash) işleminden geçirerek 0.0 ile 1.0 arası deterministik rastgele sayı üretir
fn hash3(x: i32, y: i32, z: i32, seed: i32) -> f32 {
    let mut h = seed.wrapping_add(x.wrapping_mul(374761393));
    h = h.wrapping_add(y.wrapping_mul(668265263));
    h = h.wrapping_add(z.wrapping_mul(786358481));
    h ^= h >> 13;
    h = h.wrapping_mul(236357905);
    h ^= h >> 15;
    
    (h as u32 as f32) / (u32::MAX as f32)
}

/// Lineer interpolasyon
fn lerp(a: f32, b: f32, t: f32) -> f32 {
    a + t * (b - a)
}

/// Kenar yumaşatıcı interpolasyon (Smootherstep)
fn fade(t: f32) -> f32 {
    t * t * t * (t * (t * 6.0 - 15.0) + 10.0)
}

/// 3 Boyutlu temel gürültü fonksiyonu
pub fn value_noise_3d(x: f32, y: f32, z: f32, seed: i32) -> f32 {
    let x0 = x.floor() as i32;
    let y0 = y.floor() as i32;
    let z0 = z.floor() as i32;

    let x1 = x0.wrapping_add(1);
    let y1 = y0.wrapping_add(1);
    let z1 = z0.wrapping_add(1);

    let u = fade(x - x.floor());
    let v = fade(y - y.floor());
    let w = fade(z - z.floor());

    let c000 = hash3(x0, y0, z0, seed);
    let c100 = hash3(x1, y0, z0, seed);
    let c010 = hash3(x0, y1, z0, seed);
    let c110 = hash3(x1, y1, z0, seed);
    let c001 = hash3(x0, y0, z1, seed);
    let c101 = hash3(x1, y0, z1, seed);
    let c011 = hash3(x0, y1, z1, seed);
    let c111 = hash3(x1, y1, z1, seed);

    let nx00 = lerp(c000, c100, u);
    let nx01 = lerp(c001, c101, u);
    let nx10 = lerp(c010, c110, u);
    let nx11 = lerp(c011, c111, u);

    let nxy0 = lerp(nx00, nx10, v);
    let nxy1 = lerp(nx01, nx11, v);

    lerp(nxy0, nxy1, w)
}

/// Fractal Brownian Motion (FBM) ile katmanlı organik gürültü
pub fn fbm_3d(x: f32, y: f32, z: f32, octaves: usize, seed: i32) -> f32 {
    let mut total = 0.0;
    let mut frequency = 1.0;
    let mut amplitude = 1.0;
    let mut max_value = 0.0;

    for _ in 0..octaves {
        total += value_noise_3d(x * frequency, y * frequency, z * frequency, seed) * amplitude;
        max_value += amplitude;
        amplitude *= 0.5;
        frequency *= 2.0;
    }

    total / max_value
}

/// YENİ: Noktayı belirli bir oranda sıkıştırarak (Scale) Basık Küre/Elipsoid oluşturur
pub fn sdf_ellipsoid(p: Vec3, center: Vec3, radius: f32, scale: Vec3) -> f32 {
    // Uzayı scale ile çarparak büküyoruz. (Örn: Y eksenini 2.0 ile çarparsanız, şekil Y'de basıklaşır)
    let local_p = (p - center) * scale;
    local_p.length() - radius
}

// ODALAR İÇİN (Sizin kodunuzda adı muhtemelen sdf_boxy olarak kayıtlı)
pub fn sdf_boxy(p: Vec3, center: Vec3, radius: f32, scale: Vec3) -> f32 {
    let local_p = (p - center) * scale;
    local_p.abs().max_element() - radius
}

// KORİDORLAR İÇİN (Bu eksik kalmış veya yanlış çağrılmış olabilir)
pub fn sdf_boxy_capsule(p: Vec3, a: Vec3, b: Vec3, r: f32) -> f32 {
    let pa = p - a;
    let ba = b - a;
    
    let h = (pa.dot(ba) / ba.dot(ba)).clamp(0.0, 1.0);
    let closest_point = a + ba * h;
    let d = p - closest_point;
    
    d.abs().max_element() - r
}

pub fn lerp(a: f32, b: f32, t: f32) -> f32 {
    a + t * (b - a)
}

pub fn smooth_min(a: f32, b: f32, k: f32) -> f32 {
    let h = (0.5 + 0.5 * (b - a) / k).clamp(0.0, 1.0);
    lerp(b, a, h) - k * h * (1.0 - h)
}

pub fn sdf_sphere(point: Vec3, center: Vec3, radius: f32) -> f32 {
    point.distance(center) - radius
}

/// YENİ: İki nokta arasında uzanan Kapsül (Tünel) için SDF
/// p: Uzaydaki geçerli nokta
/// a: Tünelin başlangıç noktası
/// b: Tünelin bitiş noktası
/// r: Tünelin yarıçapı
pub fn sdf_capsule(p: Vec3, a: Vec3, b: Vec3, r: f32) -> f32 {
    let pa = p - a;
    let ba = b - a;
    
    // Noktanın doğru parçası (A-B) üzerindeki izdüşüm oranını (0 ile 1 arası) buluyoruz
    let h = (pa.dot(ba) / ba.dot(ba)).clamp(0.0, 1.0);
    
    // İzdüşüm noktasına olan mesafeden yarıçapı çıkarıyoruz
    (pa - ba * h).length() - r
}

/// Herhangi bir kameraya otomatik olarak ChunkViewer ekler. 
/// Bu sayede camera kodlarına dokunmamıza gerek kalmaz.
pub fn attach_chunk_viewer_to_camera(
    mut commands: Commands,
    query: Query<Entity, (With<Camera>, Without<ChunkViewer>)>,

    for entity in query.iter() {
        commands.entity(entity).insert(ChunkViewer);
    }

pub fn chunk_tracking_system(
    viewer_query: Query<&Transform, With<ChunkViewer>>,
    config: Res<WorldGenerationConfig>,
    mut active_chunks: ResMut<ActiveChunks>,
    mut load_writer: EventWriter<ChunkLoadEvent>,
    mut unload_writer: EventWriter<ChunkUnloadEvent>,
) {
    let Ok(viewer_transform) = viewer_query.get_single() else {
        return;
    };

    let pos = viewer_transform.translation;
    let chunk_x = (pos.x / config.chunk_logical_size.x as f32).floor() as i32;
    let chunk_y = (pos.y / config.chunk_logical_size.y as f32).floor() as i32;
    let chunk_z = (pos.z / config.chunk_logical_size.z as f32).floor() as i32;
    
    let current_chunk = IVec3::new(chunk_x, chunk_y, chunk_z);

    // Eğer viewer aynı chunk içindeyse işlem yapmaya gerek yok
    if active_chunks.last_viewer_chunk == Some(current_chunk) {
        return;
    }

    active_chunks.last_viewer_chunk = Some(current_chunk);
    
    let rd = config.render_distance as i32;
    let ud = config.unload_distance as i32;

    let mut current_visible = HashSet::new();

    // Görüş mesafesindeki chunk'ları hesapla
    for x in -rd..=rd {
        for y in -rd..=rd {
            for z in -rd..=rd {
                let offset = IVec3::new(x, y, z);
                current_visible.insert(current_chunk + offset);
            }
        }
    }

    // Yüklenmesi gereken YENİ chunk'ları bul (görünür olanlar - zaten aktif olanlar)
    let chunks_to_load: Vec<IVec3> = current_visible
        .difference(&active_chunks.chunks)
        .cloned()
        .collect();

    for chunk_coord in chunks_to_load {
        active_chunks.chunks.insert(chunk_coord);
        load_writer.send(ChunkLoadEvent(chunk_coord));
        info!("ChunkLoadEvent gönderildi: {:?}", chunk_coord);
    }

    // Silinmesi gereken UZAK chunk'ları bul
    let chunks_to_unload: Vec<IVec3> = active_chunks.chunks
        .iter()
        .filter(|&&c| {
            let diff = c - current_chunk;
            diff.x.abs() > ud || diff.y.abs() > ud || diff.z.abs() > ud
        })
        .cloned()
        .collect();

    for chunk_coord in chunks_to_unload {
        active_chunks.chunks.remove(&chunk_coord);
        unload_writer.send(ChunkUnloadEvent(chunk_coord));
        info!("ChunkUnloadEvent gönderildi: {:?}", chunk_coord);
    }
}

/// Dünya koordinatından hangi X ve Z bölgesinde olduğumuzu buluruz.
pub fn get_region_coord(world_pos: Vec3, grid_size: UVec3) -> IVec2 {
    let rx = (world_pos.x / grid_size.x as f32).floor() as i32;
    let rz = (world_pos.z / grid_size.z as f32).floor() as i32;
    IVec2::new(rx, rz)
}

/// Sizin harika tam sayı bölmesi (Yuvarlama) mantığınız!
pub fn should_spawn_cave(rx: i32, rz: i32, density: f32) -> bool {
    if density <= 0.0 { return false; }
    
    // Adım 1: Tam sayı bölmesi ve yuvarlama
    let step = (1.0 / density).round() as i32;
    
    // 0.95 gibi değerler 1'e yuvarlanacağı için her bölgede çıkar
    if step <= 1 { return true; } 
    
    // Adım 2: Deterministik dağılım. (Zar atmak yerine koordinat modülünü alıyoruz 
    // ki geri döndüğümüzde mağara yer değiştirmiş olmasın).
    let hash = (rx.wrapping_mul(73856093) ^ rz.wrapping_mul(19349663)).abs();
    hash % step == 0
}

fn pick_profile_id<'a>(rules: &'a [SpawnRule], rng: &mut impl Rng) -> Option<&'a String> {
    let total_weight: f32 = rules.iter().map(|r| r.weight).sum();
    if total_weight <= 0.0 { return None; }
    
    let mut choice = rng.gen_range(0.0..total_weight);
    for rule in rules {
        choice -= rule.weight;
        if choice <= 0.0 { return Some(&rule.profile_id); }
    }
    rules.last().map(|r| &r.profile_id)
}

/// Eğer o bölgeye mağara düşüyorsa, sadece o bölge (Region) sınırları içinde Layout'u üret.
pub fn generate_region_layout(
    rx: i32,
    rz: i32,
    current_biome: &CaveTypeConfig,
    config: &CaveGenerationConfig,
) -> CaveLayout {
    // Bölge koordinatlarına göre özel seed üretimi (Aynı yere gelirsek aynı mağara çıkar)
    let seed_num = (rx.wrapping_mul(13371337) ^ rz.wrapping_mul(99991999)) as u64;
    let mut rng = StdRng::seed_from_u64(seed_num);

    let size = current_biome.grid_size;
    
    // Bu bölgenin dünya üzerindeki başlangıç (sıfır) noktası
    let origin_x = (rx as f32) * (size.x as f32);
    let origin_z = (rz as f32) * (size.z as f32);

    let mut rooms: Vec<Vec3> = Vec::new();
    let mut room_data_list: Vec<RoomData> = Vec::new();
    let target_room_count = rng.gen_range(current_biome.min_rooms..=current_biome.max_rooms);
    let surface_level = 20.0;

    for _ in 0..100 {
        if rooms.len() >= target_room_count { break; }
        
        let depth = rng.gen_range(current_biome.min_depth..=current_biome.max_depth);
        let pos_y = surface_level - depth;

        // X ve Z noktalarını kendi bölgesinin başlangıcına (origin) göre dağıtıyoruz
        let pos = Vec3::new(
            rng.gen_range((origin_x + 10.0)..(origin_x + size.x as f32 - 10.0)),
            pos_y, 
            rng.gen_range((origin_z + 10.0)..(origin_z + size.z as f32 - 10.0)),
        );
        
        if !rooms.iter().any(|&r| r.distance(pos) < 15.0) {
            rooms.push(pos);
            
            let mut r_radius = 5.0;
            let mut r_shape = ShapeType::Round;
            let mut r_scale = Vec3::ONE;
            let mut r_h_var = 0.0;
            let mut r_f_noise = 0.1;

            if let Some(profile_id) = pick_profile_id(&current_biome.room_spawn_rules, &mut rng) {
                if let Some(profile) = config.registered_rooms.iter().find(|p| &p.id == profile_id) {
                    r_radius = rng.gen_range(profile.min_radius..=profile.max_radius);
                    r_shape = profile.shape;
                    r_scale = profile.scale;
                    r_h_var = rng.gen_range(profile.min_height_variance..=profile.max_height_variance);
                    r_f_noise = profile.floor_noise;
                }
            }

            room_data_list.push(RoomData {
                id: rooms.len() - 1,
                center: pos,
                radius: r_radius,
                shape: r_shape,
                scale: r_scale,
                height_variance: r_h_var,
                floor_noise: r_f_noise,
            });
        }
    }

    let corridors_edges = calculate_mst(&rooms);
    let mut corridor_data_list = Vec::new();
    
    for edge in &corridors_edges {
        let mut c_radius = 3.0; 
        let mut c_noise_scale = current_biome.global_noise_scale;
        let mut c_floor_noise = 0.1;
        let mut c_shape = ShapeType::Round; 
        let mut c_scale = Vec3::ONE;        

        if let Some(profile_id) = pick_profile_id(&current_biome.corridor_spawn_rules, &mut rng) {
            if let Some(profile) = config.registered_corridors.iter().find(|p| &p.id == profile_id) {
                let width = rng.gen_range(profile.min_width..=profile.max_width);
                c_radius = width / 2.0; 
                c_noise_scale = profile.noise_scale;
                c_floor_noise = profile.floor_noise;
                c_shape = profile.shape;    
                c_scale = profile.scale;    
            }
        }

        corridor_data_list.push(CorridorData {
            from_room_id: edge.from,
            to_room_id: edge.to,
            radius: c_radius,
            noise_scale: c_noise_scale,
            floor_noise: c_floor_noise,
            shape: c_shape, 
            scale: c_scale, 
        });
    }

    let mut entrance_data_list = Vec::new();
    let actual_entrance_count = current_biome.entrance_count.min(rooms.len());

    for i in 0..actual_entrance_count {
        let start_room = rooms[i]; 
        let mut e_radius = 5.0;
        let mut e_ramp_length = 20.0; 

        if let Some(profile_id) = pick_profile_id(&current_biome.entrance_spawn_rules, &mut rng) {
            if let Some(profile) = config.registered_entrances.iter().find(|p| &p.id == profile_id) {
                e_radius = profile.radius;
                e_ramp_length = profile.ramp_length as f32;
            }
        }
        let end_pos = start_room + Vec3::new(e_ramp_length, (surface_level + 5.0) - start_room.y, e_ramp_length);
        entrance_data_list.push(EntranceData { start_pos: start_room, end_pos, radius: e_radius });
    }

    CaveLayout {
        rooms: room_data_list,
        corridors: corridor_data_list,
        entrances: entrance_data_list, 
        seed: seed_num as i32,
    }
}

use bevy::prelude::*;

pub fn spawn_chunk_tasks(
    mut commands: Commands,
    mut load_events: EventReader<ChunkLoadEvent>,
    config: Res<WorldGenerationConfig>,
) {
    let thread_pool = AsyncComputeTaskPool::get();

    for &ChunkLoadEvent(coord) in load_events.read() {
        let chunk_size = config.chunk_logical_size;
        
        let task = thread_pool.spawn(async move {
            let world_origin = Vec3::new(
                (coord.x * chunk_size.x as i32) as f32,
                (coord.y * chunk_size.y as i32) as f32,
                (coord.z * chunk_size.z as i32) as f32,
            );

            let mut field = ScalarField::new(coord, chunk_size, world_origin);
            
            for z in 0..field.padded_size.z {
                for y in 0..field.padded_size.y {
                    for x in 0..field.padded_size.x {
                        let global_pos = world_origin + Vec3::new(x as f32, y as f32, z as f32);
                        let wave = (global_pos.x * 0.1).sin() * 5.0 + (global_pos.z * 0.1).cos() * 5.0;
                        let density = global_pos.y - wave; 
                        
                        field.set_density(x, y, z, density);
                    }
                }
            }

            ChunkResult {
                coord,
                field,
                mesh: None, 
                generated_lights: Vec::new(),
            }
        });

        // YENİ: Task'ı Option içine Some ile yerleştiriyoruz
        commands.spawn((
            ChunkGenTask(Some(task)),
            ChunkEntity(coord),
        ));
    }
}

pub fn handle_completed_chunk_tasks(
    mut commands: Commands,
    mut chunk_manager: ResMut<ChunkManager>,
    mut task_query: Query<(Entity, &mut ChunkGenTask)>,
) {
    for (entity, mut task_component) in task_query.iter_mut() {
        // YENİ: Task'ı Option içinden alıyoruz. Eğer daha önce tamamlanıp
        // None olduysa bu blok hiç çalışmayacak ve panik yaşanmayacak.
        if let Some(mut task) = task_component.0.take() {
            if let Some(result) = block_on(future::poll_once(&mut task)) {
                
                chunk_manager.chunks.insert(result.coord, result.field);
                info!("Chunk üretimi başarıyla tamamlandı: {:?}", result.coord);

                commands.entity(entity).remove::<ChunkGenTask>();
            } else {
                // İşlem henüz bitmemiş. Task'ı Option içine geri iade ediyoruz.
                task_component.0 = Some(task);
            }
        }
    }
}

pub fn handle_chunk_unloads(
    mut commands: Commands,
    mut unload_events: EventReader<ChunkUnloadEvent>,
    mut chunk_manager: ResMut<ChunkManager>,
    chunk_entities: Query<(Entity, &ChunkEntity)>,
) {
    for &ChunkUnloadEvent(coord) in unload_events.read() {
        chunk_manager.chunks.remove(&coord);
        
        for (entity, chunk_entity) in chunk_entities.iter() {
            if chunk_entity.0 == coord {
                commands.entity(entity).despawn_recursive();
                info!("Chunk bellekten ve sahneden temizlendi: {:?}", coord);
                break;
            }
        }
    }
}

pub enum ZoneType {
    Room(usize),
    Corridor(usize, usize),
    Wilderness,
}

#[derive(Resource)]
pub struct CaveLayout {
    pub rooms: Vec<RoomData>,
    pub corridors: Vec<CorridorData>,
    pub entrances: Vec<EntranceData>, // YENİ: Giriş rampaları
    pub seed: i32, // YENİ: Mağaranın kimliği olan seed artık Layout içinde saklanıyor

impl CaveLayout {
    pub fn get_zone_at(&self, point: Vec3) -> ZoneType {
        for room in &self.rooms {
            if point.distance_squared(room.center) <= (room.radius * room.radius) {
                return ZoneType::Room(room.id);
            }
        }

        for corridor in &self.corridors {
            let p_a = self.rooms[corridor.from_room_id].center;
            let p_b = self.rooms[corridor.to_room_id].center;
            
            let pa = point - p_a;
            let ba = p_b - p_a;
            let h = (pa.dot(ba) / ba.dot(ba)).clamp(0.0, 1.0);
            let closest_point = p_a + ba * h;
            
            if point.distance_squared(closest_point) <= (corridor.radius * corridor.radius) {
                return ZoneType::Corridor(corridor.from_room_id, corridor.to_room_id);
            }
        }

        ZoneType::Wilderness
    }
}

struct UnionFind {
    parent: Vec<usize>,
}

impl UnionFind {
    fn new(size: usize) -> Self {
        Self {
            parent: (0..size).collect(),
        }
    }

    fn find(&mut self, i: usize) -> usize {
        if self.parent[i] == i {
            i
        } else {
            let root = self.find(self.parent[i]);
            self.parent[i] = root;
            root
        }
    }

    fn union(&mut self, i: usize, j: usize) -> bool {
        let root_i = self.find(i);
        let root_j = self.find(j);
        
        if root_i != root_j {
            self.parent[root_i] = root_j;
            true
        } else {
            false
        }
    }
}

pub fn calculate_mst(rooms: &[Vec3]) -> Vec<Edge> {
    let mut edges = Vec::new();
    let num_rooms = rooms.len();

    for i in 0..num_rooms {
        for j in (i + 1)..num_rooms {
            let dist = rooms[i].distance(rooms[j]);
            edges.push(Edge { from: i, to: j, distance: dist });
        }
    }

    edges.sort_by(|a, b| a.distance.partial_cmp(&b.distance).unwrap());
    
    let mut mst = Vec::new();
    let mut uf = UnionFind::new(num_rooms);
    
    for edge in edges {
        if uf.union(edge.from, edge.to) {
            mst.push(edge);
            if mst.len() == num_rooms - 1 {
                break;
            }
        }
    }

    mst
}

        let chunk_x = (local_to_grid.x / chunk_size.x as f32).floor() as i32;
        let chunk_y = (local_to_grid.y / chunk_size.y as f32).floor() as i32;
        let chunk_z = (local_to_grid.z / chunk_size.z as f32).floor() as i32;
        let chunk_coord = IVec3::new(chunk_x, chunk_y, chunk_z);

        if let Some(field) = self.chunks.get(&chunk_coord) {

            let x = local_pos.x.round() as u32;
            let y = local_pos.y.round() as u32;
            let z = local_pos.z.round() as u32;

            if x < field.padded_size.x && y < field.padded_size.y && z < field.padded_size.z {
                return field.get_light(x, y, z);
            }
        }

        Vec3::ZERO 
    }

#[derive(Clone)]
pub struct ScalarField {
    pub chunk_coord: IVec3,
    pub logical_size: UVec3,
    pub padded_size: UVec3,
    pub data: Vec<f32>,
    pub light_data: Vec<Vec3>,
    pub world_origin: Vec3,
}

impl ScalarField {
    pub fn new(chunk_coord: IVec3, logical_size: UVec3, world_origin: Vec3) -> Self {
        let padded_size = UVec3::new(logical_size.x + 1, logical_size.y + 1, logical_size.z + 1);
        let total_cells = (padded_size.x * padded_size.y * padded_size.z) as usize;
        Self {
            chunk_coord,
            logical_size,
            padded_size,
            data: vec![0.0; total_cells],
            // DÜZELTME: Zifiri karanlık (ZERO) yerine loş bir mağara mavisi veriyoruz.
            // Böylece ışık ulaşmayan chunk'lar simsiyah kalmayacak.
            light_data: vec![Vec3::new(4.95, 4.95, 8.08); total_cells], 
            world_origin,
        }
    }

    pub fn get_index(&self, x: u32, y: u32, z: u32) -> usize {
        (z * self.padded_size.x * self.padded_size.y + y * self.padded_size.x + x) as usize
    }

    pub fn get_density(&self, x: u32, y: u32, z: u32) -> f32 {
        if x >= self.padded_size.x || y >= self.padded_size.y || z >= self.padded_size.z {
            return 0.0;
        }
        self.data[self.get_index(x, y, z)]
    }

    pub fn set_density(&mut self, x: u32, y: u32, z: u32, value: f32) {
        if x < self.padded_size.x && y < self.padded_size.y && z < self.padded_size.z {
            let idx = self.get_index(x, y, z);
            self.data[idx] = value;
        }
    }

    pub fn get_light(&self, x: u32, y: u32, z: u32) -> Vec3 {
        if x >= self.padded_size.x || y >= self.padded_size.y || z >= self.padded_size.z { return Vec3::ZERO; }
        self.light_data[self.get_index(x, y, z)]
    }

    pub fn set_light(&mut self, x: u32, y: u32, z: u32, color: Vec3) {
        if x < self.padded_size.x && y < self.padded_size.y && z < self.padded_size.z {
            let idx = self.get_index(x, y, z);
            self.light_data[idx] = color;
        }
    }

    pub fn compute_lighting(&mut self, world_light_pos: Vec3, color: Vec3, max_distance: i32) {
        let local_pos = world_light_pos - self.world_origin;
        let lx = local_pos.x.round() as i32;
        let ly = local_pos.y.round() as i32;
        let lz = local_pos.z.round() as i32;

        if lx < 0 || ly < 0 || lz < 0 || lx >= self.padded_size.x as i32 || ly >= self.padded_size.y as i32 || lz >= self.padded_size.z as i32 {
            return;
        }

        let mut queue = VecDeque::new();
        queue.push_back((lx as u32, ly as u32, lz as u32, max_distance));
        self.set_light(lx as u32, ly as u32, lz as u32, color);

        let falloff = 1.0 / (max_distance as f32);

        while let Some((x, y, z, dist)) = queue.pop_front() {
            if dist <= 0 { continue; }

            let neighbors = [
                (x.wrapping_sub(1), y, z), (x + 1, y, z),
                (x, y.wrapping_sub(1), z), (x, y + 1, z),
                (x, y, z.wrapping_sub(1)), (x, y, z + 1),
            ];

            let current_light = self.get_light(x, y, z);

            for &(nx, ny, nz) in &neighbors {
                if nx < self.padded_size.x && ny < self.padded_size.y && nz < self.padded_size.z {
                    let density = self.get_density(nx, ny, nz);
                    let target_color = current_light - (color * falloff); 
                    let target_color = target_color.max(Vec3::ZERO);
                    let existing_light = self.get_light(nx, ny, nz);

                    if target_color.length_squared() > existing_light.length_squared() + 0.01 {
                        self.set_light(nx, ny, nz, target_color);
                        if density < 0.35 {
                            queue.push_back((nx, ny, nz, dist - 1));
                        }
                    }
                }
            }
        }
    }
}

                                }
                            }
                            check_y -= 1.0;
                        }
                        r.center.y // Bulamazsa fallback
                    };

                    true_floor_y = get_ground_y(r.center.x, r.center.z);

                    // Eğim: Odanın merkezinden kenarlarına (radius * 0.8) doğru zemin yüksekliğindeki değişimi ölç
                    let offset = r.radius * 0.8;
                    let left_y = get_ground_y(r.center.x + offset, r.center.z);
                    let right_y = get_ground_y(r.center.x - offset, r.center.z);
                    let up_y = get_ground_y(r.center.x, r.center.z + offset);
                    let down_y = get_ground_y(r.center.x, r.center.z - offset);

                    let diff_x = (left_y - right_y).abs();
                    let diff_z = (up_y - down_y).abs();
                    average_slope = diff_x.max(diff_z); // Yükseklik farkı (Eğim)
                }
                
                rooms.push(RoomAnalysis {
                    id: r.id,
                    center: r.center,
                    radius: r.radius,
                    bounds_min: r.center - extents,
                    bounds_max: r.center + extents,
                    is_dead_end: connection_counts[r.id] == 1,
                    max_clearance,
                    average_slope,
                    true_floor_y,
                });
            }

            let mut corridors = Vec::new();
            for c in &layout.corridors {
                corridors.push(CorridorAnalysis {
                    from_room_id: c.from_room_id,
                    to_room_id: c.to_room_id,
                    radius: c.radius,
                });
            }

            let mut entrances = Vec::new();
            for e in &layout.entrances {
                entrances.push(EntranceAnalysis {
                    start_pos: e.start_pos,
                    end_pos: e.end_pos,
                    radius: e.radius,
                });
            }

            commands.spawn(RegionAnalysisData {
                region_coord,
                rooms,
                corridors,
                entrances,
                sockets: Vec::new(),
                chokepoints,
            });

            tracker.processed_regions.insert(region_coord);
        }
    }
}
