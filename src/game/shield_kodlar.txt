use bevy::prelude::*;
use super::events::{BlockEvent, RecoilEvent};
use super::systems::{handle_shield_input, update_shield_orientation};

pub struct ShieldPlugin;

impl Plugin for ShieldPlugin {
    fn build(&self, app: &mut App) {
        app.add_event::<RecoilEvent>();
        app.add_event::<BlockEvent>();
        
        // Sistemleri Update döngüsüne ekliyoruz
        app.add_systems(Update, (
            handle_shield_input,
            update_shield_orientation,
        ).chain()); // Sıralı çalışması veri bütünlüğü için iyidir
    }
}

use bevy::input::mouse::MouseMotion;
use bevy::prelude::*;

use super::components::{BlockingState, EquippedShield, Shield};

// 1. Kalkan Durum Makinesi (State Machine)
// Sağ tıka basıldığında Parry penceresini başlatır, süre dolunca normal Block durumuna geçer.
pub fn handle_shield_input(
    mouse: Res<ButtonInput<MouseButton>>,
    time: Res<Time>,
    mut query: Query<(&mut BlockingState, &Shield)>,
) {
    for (mut state, shield) in query.iter_mut() {
        if mouse.just_pressed(MouseButton::Right) {
            // Sadece Idle durumundayken kalkan kaldırılabilir
            if matches!(*state, BlockingState::Idle) {
                *state = BlockingState::Parrying {
                    timer: Timer::from_seconds(shield.parry_window, TimerMode::Once),
                };
            }
        } else if mouse.pressed(MouseButton::Right) {
            // Sağ tıka basılı tutuluyorsa Parry zamanlayıcısını ilerlet
            if let BlockingState::Parrying { timer } = &mut *state {
                timer.tick(time.delta());
                
                // Parry süresi dolduysa normal bloklamaya geç
                if timer.just_finished() {
                    *state = BlockingState::Blocking;
                    println!("Parry süresi bitti, normal Block durumuna geçildi.");
                }
            }
        } else if mouse.just_released(MouseButton::Right) {
            // Sağ tık bırakıldığında kalkanı indir (GuardBroken değilse)
            if matches!(*state, BlockingState::Parrying { .. } | BlockingState::Blocking) {
                *state = BlockingState::Idle;
            }
        }
    }
}

// 2. Yönlü Savunma (Directional Blocking)
// Fare hareketlerini okuyarak kalkanın fiziksel collider'ını karakter etrafında eğer/döndürür.
pub fn update_shield_orientation(
    mut motion_evr: EventReader<MouseMotion>,
    mouse: Res<ButtonInput<MouseButton>>,
    mut query: Query<&mut Transform, With<EquippedShield>>,
) {
    // Sadece sağ tıka basılıyken kalkan yönlendirilebilir
    if !mouse.pressed(MouseButton::Right) {
        // İstersen burada kalkan transformunu lerp ile yavaşça varsayılan (merkez) pozisyona çekebilirsin.
        for _ in motion_evr.read() {} // Birikmiş eventleri tüket
        return;
    }

    let mut total_delta = Vec2::ZERO;
    for ev in motion_evr.read() {
        total_delta += ev.delta;
    }

    if total_delta != Vec2::ZERO {
        let sensitivity = 0.005; // Fare hassasiyeti
        
        for mut transform in query.iter_mut() {
            let yaw = -total_delta.x * sensitivity;
            let pitch = -total_delta.y * sensitivity;

            // Kalkanı farenin hareketine göre döndür
            transform.rotate_local_y(yaw);
            transform.rotate_local_x(pitch);
            
            // NOT: İleride bu transform değerlerini clamp'leyerek (sınırlandırarak) 
            // kalkanın karakterin kafasının arkasına kadar dönmesini engelleyebilirsin.
        }
    }
}
