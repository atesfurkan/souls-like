/// Karakterin belirli bir uzvu için ulaşılması hedeflenen noktayı temsil eder.
/// Bu bileşen genellikle oyundaki mantık (Combat/Shield) tarafından güncellenir.
#[derive(Component, Debug, Reflect)]
pub struct IkTarget {
    pub position: Vec3, // Hedefin dünya veya lokal koordinattaki pozisyonu
    /// 0.0 (Tamamen FK/Animasyon) ile 1.0 (Tamamen IK) arası geçiş değeri.
    /// Smooth interpolation (slerp) için kullanılır.
    pub weight: f32, 
}

/// 2-Bone IK algoritmasının ihtiyaç duyduğu kemik zinciri.
/// (Örn: UpperArm -> Forearm -> Hand)
#[derive(Component, Debug, Reflect)]
pub struct TwoBoneIkChain {
    pub root: Entity, // Kök kemik (Örn: Omuz / LeftArm)
    pub mid: Entity,  // Orta kemik (Örn: Dirsek / LeftForearm)
    pub tip: Entity,  // Uç kemik (Örn: El / LeftHand)
    pub pole_vector: Vec3, // Dirseğin veya dizin bükülme yönünü belirleyen vektör
}

pub fn update_shield_ik_target(
    // Sorguya 'mut TwoBoneIkChain' eklendi!
    mut query: Query<(&mut IkTarget, &mut TwoBoneIkChain, &GuardIntent, &BlockingState, &Transform), With<Player>>,
    lock_on: Res<LockOnState>,
    time: Res<Time>,
) {
    for (mut ik_target, mut chain, intent, state, transform) in query.iter_mut() {
        let is_guarding = lock_on.target.is_some() && matches!(state, BlockingState::Blocking);
        
        let target_w = if is_guarding { 1.0 } else { 0.0 };
        ik_target.weight = ik_target.weight.lerp(target_w, time.delta_seconds() * 15.0);

        if ik_target.weight > 0.01 {
            // 1. Kalkan Hedefi (Elin gideceği nokta)
            let local_base_pos = Vec3::new(0.2, 1.2, 0.4); 
            let local_target = local_base_pos + Vec3::new(intent.offset.x, intent.offset.y, 0.0);
            ik_target.position = transform.transform_point(local_target);
            
            // 2. ÇÖZÜM: Pole Vector (Dirseğin bakacağı nokta)
            // Dirseğin oyuncunun soluna (-1.0), aşağısına (0.5) ve biraz gerisine (-0.5) bakmasını istiyoruz.
            let local_pole = Vec3::new(-1.0, 0.5, -0.5);
            // Bu lokal noktayı, oyuncunun dünyadaki konumuna çevirip zincire veriyoruz!
            chain.pole_vector = transform.transform_point(local_pole);
        }
    }
}


// Hiyerarşiyi tırmanarak 1-frame gecikmesiz anlık Global Transform hesaplayan yardımcı fonksiyon
fn compute_manual_global_transform(
    entity: Entity,
    transforms: &Query<&mut Transform>,
    parents: &Query<&Parent>,
) -> Transform {
    let mut current_entity = entity;
    let mut global_transform = Transform::IDENTITY;
    let mut hierarchy = Vec::new();

    while let Ok(transform) = transforms.get(current_entity) {
        hierarchy.push(*transform);
        if let Ok(parent) = parents.get(current_entity) {
            current_entity = parent.get();
        } else {
            break;
        }
    }

    for transform in hierarchy.into_iter().rev() {
        global_transform = global_transform * transform;
    }
    global_transform
}

pub fn solve_two_bone_ik(
    chain_query: Query<(&TwoBoneIkChain, &IkTarget)>,
    mut transforms: Query<&mut Transform>,
    parent_query: Query<&Parent>,
) {
    for (chain, target) in chain_query.iter() {
        if target.weight <= 0.0 { continue; }

        // 1. KEMİK UZUNLUKLARINI LOKAL UZAYDAN ÇEKME
        // Dirsek omuzun çocuğu, el dirseğin çocuğu olduğu için lokal translation uzunlukları doğrudan kemik boyudur.
        let len1 = transforms.get(chain.mid).map(|t| t.translation.length()).unwrap_or(0.001);
        let len2 = transforms.get(chain.tip).map(|t| t.translation.length()).unwrap_or(0.001);
        let total_len = len1 + len2;

        // 2. OMUZUN ANLIK DÜNYA UZAYI (Jitter-Free)
        let root_parent = parent_query.get(chain.root).map(|p| p.get()).unwrap_or(chain.root);
        let root_parent_global = compute_manual_global_transform(root_parent, &transforms, &parent_query);
        
        let root_local = transforms.get(chain.root).cloned().unwrap_or_default();
        let root_global_pos = (root_parent_global * root_local).translation;

        // 3. HEDEFE YÖNELİK SOFT IK HESAPLAMASI
        let mut dir_target = target.position - root_global_pos;
        let mut dist_target = dir_target.length();
        let dir_target_norm = dir_target.normalize_or_zero();

        let d_max = total_len * 0.95;
        
        if dist_target > d_max {
            // Soft IK Formülü uygulaması
            let diff = dist_target - d_max;
            let max_extend = total_len - d_max;
            dist_target = d_max + max_extend * (1.0 - (-diff / max_extend).exp());
        } else {
            dist_target = dist_target.max(0.001); // 0'a bölme hatasını engelle
        }

        // 4. KUTUPSAL VEKTÖR (POLE VECTOR) DÖNÜŞÜMÜ
        // Pole Vector'u karakterin (root_parent) dönüş yönüne göre hizala
        let global_pole_vector = root_parent_global.rotation * chain.pole_vector;
        let mut bend_normal = dir_target_norm.cross(global_pole_vector).normalize_or_zero();
        
        if bend_normal == Vec3::ZERO {
            bend_normal = dir_target_norm.cross(Vec3::Y).normalize_or_zero();
        }

        // 5. İÇ AÇI (KOSİNÜS TEOREMİ) VE YENİ POZİSYONLAR
        let cos_angle_root = ((len1 * len1) + (dist_target * dist_target) - (len2 * len2)) / (2.0 * len1 * dist_target);
        let angle_root = cos_angle_root.clamp(-1.0, 1.0).acos();

        let new_mid_dir = Quat::from_axis_angle(bend_normal, angle_root) * dir_target_norm;
        
        let root_global_rot = (root_parent_global * root_local).rotation;
        
        // Varsayılan kolların uzanma eksenini (Forward) belirliyoruz. Mixamo riglerinde genellikle Y veya -Y eksenidir.
        let default_bone_dir = transforms.get(chain.mid).map(|t| t.translation.normalize_or_zero()).unwrap_or(Vec3::Y);
        let current_mid_global_dir = root_global_rot * default_bone_dir;

        // 6. OMUZ VE DİRSEK LOKAL ROTASYON YAZIMI
        let root_rot_diff = Quat::from_rotation_arc(current_mid_global_dir, new_mid_dir);
        let target_root_global_rot = root_rot_diff * root_global_rot;
        
        // Global rotasyonu Root'un ebeveynine göre Lokal'e çeviriyoruz
        let target_root_local_rot = root_parent_global.rotation.inverse() * target_root_global_rot;

        if let Ok(mut root_transform) = transforms.get_mut(chain.root) {
            root_transform.rotation = root_transform.rotation.slerp(target_root_local_rot, target.weight);
        }

        // Dirsek (Mid) dönüşü lokal eksende bükülme açısına eşittir.
        // Omuz zaten doğru yöne döndüğü için, dirseği sadece 'bend_normal' etrafında bükmek yeterlidir.
        let cos_angle_mid = ((len1 * len1) + (len2 * len2) - (dist_target * dist_target)) / (2.0 * len1 * len2);
        let angle_mid = cos_angle_mid.clamp(-1.0, 1.0).acos();
        
        // Düz kol açısı PI'dir, bükülme miktarı PI - angle_mid
        let bend_amount = std::f32::consts::PI - angle_mid;
        let mid_local_rot = Quat::from_axis_angle(Vec3::X, bend_amount); // Mixamo'da diz/dirsek bükülme ekseni X veya Z olabilir, rig'e göre test edilmeli.

        if let Ok(mut mid_transform) = transforms.get_mut(chain.mid) {
             let base_rot = Quat::IDENTITY; // FK animasyonunu korumak istiyorsak mid_transform.rotation saklanabilir.
             mid_transform.rotation = base_rot.slerp(mid_local_rot, target.weight);
        }
    }
}

use bevy::prelude::*;
use super::components::{CharacterBones, CharacterAnimations,

pub const BONE_LEFT_ARM: &str = "mixamorig:LeftArm"; // YENİ

pub fn update_animations_from_intent(
    mut anim_query: Query<(
        Entity, // &Parent yerine Entity alıyoruz
        &AnimationNodes, 
        &mut ActiveAnimation, 
        &mut AnimationPlayer
    )>,
    parent_query: Query<&Parent>, // Hiyerarşide yukarı tırmanmak için
    intent_query: Query<&AnimationIntent>,
) {
    for (entity, nodes, mut active_anim, mut player) in anim_query.iter_mut() {
        
        // --- HİYERARŞİ TIRMANIŞI ---
        let mut current_entity = entity;
        let mut found_intent = None;
        
        // En üste (kapsüle) ulaşana kadar ebeveynleri kontrol et
        while let Ok(parent) = parent_query.get(current_entity) {
            if let Ok(intent) = intent_query.get(parent.get()) {
                found_intent = Some(intent);
                break; // Niyeti bulduk, tırmanmayı bırak
            }
            current_entity = parent.get(); // Bulamadıysak bir üst ebeveyne geç
        }

        // Eğer kapsüle kadar çıkıp hala AnimationIntent bulamadıysa pas geç
        let Some(intent) = found_intent else {
            continue;
        };
        // ---------------------------

        // YENİ MANTIK: Eğer havadaysak tek öncelik zıplamadır.
        let target_node = if intent.is_in_air {
            nodes.jump
        } else if intent.is_moving {
            if intent.is_sprinting {
                nodes.run
            } else {
                nodes.walk
            }
        } else {
            nodes.idle
        };

        if active_anim.0 == target_node {
            continue;
        }

        player.stop_all();
        
        // Zıplama animasyonu sonsuza kadar kendini tekrar etmesin (.repeat() YOK)
        if target_node == nodes.jump {
            player.play(target_node);
        } else {
            player.play(target_node).repeat();
        }
        
        active_anim.0 = target_node;
    }
}

pub fn load_animations(
    mut commands: Commands,
    asset_server: Res<AssetServer>,
) {
    // .glb dosyalarının içindeki ilk animasyon klibine (Animation0) ulaşıyoruz
    let animations = CharacterAnimations {
        idle: asset_server.load("models/animations/idle.glb#Animation0"),
        walk: asset_server.load("models/animations/walk_sword_shield.glb#Animation0"),
        run: asset_server.load("models/animations/run_sword_shield.glb#Animation0"),
        jump: asset_server.load("models/animations/jump_sword_shield.glb#Animation0"),
    };
    
    commands.insert_resource(animations);
    info!("Karakter animasyonları belleğe yüklendi.");
}



pub fn setup_animation_graph(
    mut commands: Commands,
    mut players: Query<(Entity, &mut AnimationPlayer), Added<AnimationPlayer>>,
    animations: Res<CharacterAnimations>,
    mut graphs: ResMut<Assets<AnimationGraph>>,
) {
    for (entity, mut player) in &mut players {
        let mut graph = AnimationGraph::new();
        
        let idle_node = graph.add_clip(animations.idle.clone(), 1.0, graph.root);
        let walk_node = graph.add_clip(animations.walk.clone(), 1.0, graph.root);
        let run_node = graph.add_clip(animations.run.clone(), 1.0, graph.root);
        let jump_node = graph.add_clip(animations.jump.clone(), 1.0, graph.root); // YENİ
        
        let graph_handle = graphs.add(graph);
        commands.entity(entity).insert(graph_handle);
        
        commands.entity(entity).insert(AnimationNodes {
            idle: idle_node,
            walk: walk_node,
            run: run_node,
            jump: jump_node, // YENİ
        });
        
        player.play(idle_node).repeat();
        commands.entity(entity).insert(ActiveAnimation(idle_node));
        
        info!("AnimationGraph tüm kliplerle kuruldu ve Idle başlatıldı.");
    }
}

pub fn cache_character_bones(
    mut query: Query<(Entity, &mut CharacterBones)>,
    children_query: Query<&Children>,
    names_query: Query<(Entity, &Name)>,
) {
    for (entity, mut bones) in query.iter_mut() {
        if bones.is_linked { continue; }

        let mut found_count = 0;
        let target_count = 4; // Kalkan zinciri(3) + Silah kemiği(1)[cite: 3]

        for descendant in children_query.iter_descendants(entity) {
            if let Ok((bone_entity, name)) = names_query.get(descendant) {
                let name_str = name.as_str();
                
                match name_str {
                    BONE_RIGHT_HAND => {
                        bones.right_hand = Some(bone_entity);
                        found_count += 1;
                    },
                    BONE_LEFT_ARM => {
                        bones.left_arm = Some(bone_entity);
                        found_count += 1;
                    },
                    BONE_LEFT_FOREARM => {
                        bones.left_forearm = Some(bone_entity);
                        found_count += 1;
                    },
                    BONE_LEFT_HAND => {
                        bones.left_hand = Some(bone_entity);
                        found_count += 1;
                    },
                    _ => {}
                }
            }
        }

        // Gerekli kemikler bulunduysa sistemi işaretle[cite: 3]
        if found_count >= target_count {
            bones.is_linked = true;
            info!("Kemikler başarıyla tarandı ve IK önbelleğine alındı!"); //[cite: 3]
        }
    }
}
